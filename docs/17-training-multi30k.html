이제 진짜로 학습한다. 지금까지 만든 것이 모두 여기 모인다.

train_worker: GPU 하나에서 도는 학습 과정이다. 1. make_model로 6층 모델을 만들어 GPU에 올린다(분산 학습이면 DDP로 감싼다). 2. LabelSmoothing(smoothing=0.1), create_dataloaders, Adam(β=(0.9, 0.98), ε=1e-9) + rate 스케줄을 준비한다. 3. epoch마다 run_epoch로 학습하고 체크포인트를 저장한 뒤 검증 손실을 잰다.

train_distributed_model/train_model: GPU가 여러 개면 mp.spawn으로 GPU마다 train_worker를 하나씩 띄운다.

load_trained_model: 학습 설정이 여기 있다. 배치 32, accum_iter=10(10배치마다 한 번 업데이트하므로 실질 배치 320), warmup 3000, 8 epoch. multi30k_model_final.pt가 이미 있으면 학습을 건너뛴다.

check_outputs: 검증 문장을 greedy_decode로 번역해서 원문·정답·모델 출력을 나란히 찍는다.

21def attention(query, key, value, mask=None, dropout=None):
22 "Compute 'Scaled Dot Product Attention'"

29 p_attn = dropout(p_attn)

47class MultiHeadedAttention(nn.Module):
48 def __init__(self, h, d_model, dropout=0.1):
49 "Take in model size and number of heads."
50 super(MultiHeadedAttention, self).__init__()
51 assert d_model % h == 0
52 # We assume d_v always equals d_k
53 self.d_k = d_model // h
54 self.h = h
55 self.linears = clones(nn.Linear(d_model, d_model), 4)
56 self.attn = None
57 self.dropout = nn.Dropout(p=dropout)
58
59 def forward(self, query, key, value, mask=None):
60 "Implements Figure 2"
61 if mask is not None:
62 # Same mask applied to all h heads.
63 mask = mask.unsqueeze(1)
64 nbatches = query.size(0)
65
66 # 1) Do all the linear projections in batch from d_model => h x d_k
67 query, key, value = [
68 lin(x).view(nbatches, -1, self.h, self.d_k).transpose(1, 2)
69 for lin, x in zip(self.linears, (query, key, value))
70 ]
71
72 # 2) Apply attention on all the projected vectors in batch.
73 x, self.attn = attention(
74 query, key, value, mask=mask, dropout=self.dropout
75 )
76
77 # 3) "Concat" using a view and apply a final linear.
78 x = (
79 x.transpose(1, 2)
80 .contiguous()
81 .view(nbatches, -1, self.h * self.d_k)
82 )
83 del query
84 del key
85 del value
86 return self.linears[-1](x)

210 m = memory
211 x = self.sublayer[0](x, lambda x: self.self_attn(x, x, x, tgt_mask))
212 x = self.sublayer[1](x, lambda x: self.src_attn(x, m, m, src_mask))
213 return self.sublayer[2](x, self.feed_forward)

216class Decoder(nn.Module):
217 "Generic N layer decoder with masking."
218
219 def __init__(self, layer, N):
220 super(Decoder, self).__init__()
221 self.layers = clones(layer, N)
222 self.norm = LayerNorm(layer.size)
223
224 def forward(self, x, memory, src_mask, tgt_mask):
225 for layer in self.layers:
226 x = layer(x, memory, src_mask, tgt_mask)
227 return self.norm(x)

230class EncoderDecoder(nn.Module):
231 """
232 A standard Encoder-Decoder architecture. Base for this and many
233 other models.
234 """
235
236 def __init__(self, encoder, decoder, src_embed, tgt_embed, generator):
237 super(EncoderDecoder, self).__init__()
238 self.encoder = encoder
239 self.decoder = decoder
240 self.src_embed = src_embed
241 self.tgt_embed = tgt_embed
242 self.generator = generator
243
244 def forward(self, src, tgt, src_mask, tgt_mask):
245 "Take in and process masked src and target sequences."
246 return self.decode(self.encode(src, src_mask), src_mask, tgt, tgt_mask)
247
248 def encode(self, src, src_mask):
249 return self.encoder(self.src_embed(src), src_mask)
250
251 def decode(self, memory, src_mask, tgt, tgt_mask):
252 return self.decoder(self.tgt_embed(tgt), memory, src_mask, tgt_mask)

256 "Define standard linear + softmax generation step."

259 super(Generator, self).__init__()

266def make_model(
267 src_vocab, tgt_vocab, N=6, d_model=512, d_ff=2048, h=8, dropout=0.1
268):
269 "Helper: Construct a model from hyperparameters."
270 c = copy.deepcopy
271 attn = MultiHeadedAttention(h, d_model)
272 ff = PositionwiseFeedForward(d_model, d_ff, dropout)
273 position = PositionalEncoding(d_model, dropout)
274 model = EncoderDecoder(
275 Encoder(EncoderLayer(d_model, c(attn), c(ff), dropout), N),
276 Decoder(DecoderLayer(d_model, c(attn), c(attn), c(ff), dropout), N),
277 nn.Sequential(Embeddings(d_model, src_vocab), c(position)),
278 nn.Sequential(Embeddings(d_model, tgt_vocab), c(position)),
279 Generator(d_model, tgt_vocab),
280 )
281
282 # This was important from their code.
283 # Initialize parameters with Glorot / fan_avg.
284 for p in model.parameters():
285 if p.dim() > 1:
286 nn.init.xavier_uniform_(p)
287 return model

290def greedy_decode(model, src, src_mask, max_len, start_symbol):
291 memory = model.encode(src, src_mask)
292 ys = torch.zeros(1, 1).fill_(start_symbol).type_as(src.data)
293 for i in range(max_len - 1):
294 out = model.decode(
295 memory, src_mask, ys, subsequent_mask(ys.size(1)).type_as(src.data)
296 )
297 prob = model.generator(out[:, -1])
298 _, next_word = torch.max(prob, dim=1)
299 next_word = next_word.data[0]
300 ys = torch.cat(
301 [ys, torch.zeros(1, 1).type_as(src.data).fill_(next_word)], dim=1
302 )
303 return ys

306class Batch:
307 """Object for holding a batch of data with mask during training."""
308
309 def __init__(self, src, tgt=None, pad=2): # 2 = <blank>
310 self.src = src
311 self.src_mask = (src != pad).unsqueeze(-2)
312 if tgt is not None:
313 self.tgt = tgt[:, :-1]
314 self.tgt_y = tgt[:, 1:]
315 self.tgt_mask = self.make_std_mask(self.tgt, pad)
316 self.ntokens = (self.tgt_y != pad).data.sum()
317
318 @staticmethod
319 def make_std_mask(tgt, pad):
320 "Create a mask to hide padding and future words."
321 tgt_mask = (tgt != pad).unsqueeze(-2)
322 tgt_mask = tgt_mask & subsequent_mask(tgt.size(-1)).type_as(
323 tgt_mask.data
324 )
325 return tgt_mask

328class LabelSmoothing(nn.Module):
329 "Implement label smoothing."
330
331 def __init__(self, size, padding_idx, smoothing=0.0):
332 super(LabelSmoothing, self).__init__()
333 self.criterion = nn.KLDivLoss(reduction="sum")
334 self.padding_idx = padding_idx
335 self.confidence = 1.0 - smoothing
336 self.smoothing = smoothing
337 self.size = size
338 self.true_dist = None
339
340 def forward(self, x, target):
341 assert x.size(1) == self.size
342 true_dist = x.data.clone()
343 true_dist.fill_(self.smoothing / (self.size - 2))
344 true_dist.scatter_(1, target.data.unsqueeze(1), self.confidence)
345 true_dist[:, self.padding_idx] = 0
346 mask = torch.nonzero(target.data == self.padding_idx)
347 if mask.dim() > 0:
348 true_dist.index_fill_(0, mask.squeeze(), 0.0)
349 self.true_dist = true_dist
350 return self.criterion(x, true_dist.clone().detach())

353class SimpleLossCompute:
354 "A simple loss compute and train function."
355
356 def __init__(self, generator, criterion):
357 self.generator = generator
358 self.criterion = criterion
359
360 def __call__(self, x, y, norm):
361 x = self.generator(x)
362 sloss = (
363 self.criterion(
364 x.contiguous().view(-1, x.size(-1)), y.contiguous().view(-1)
365 )
366 / norm
367 )
368 return sloss.data * norm, sloss

371def rate(step, model_size, factor, warmup):
372 """
373 we have to default the step to 1 for LambdaLR function
374 to avoid zero raising to negative power.
375 """
376 if step == 0:
377 step = 1
378 return factor * (
379 model_size ** (-0.5) * min(step ** (-0.5), step * warmup ** (-1.5))

384 """Track number of steps, examples, and tokens processed"""

386 step: int = 0 # Steps in the current epoch

392def run_epoch(
393 data_iter,
394 model,
395 loss_compute,
396 optimizer,
397 scheduler,
398 mode="train",
399 accum_iter=1,
400 train_state=TrainState(),
401):
402 """Train a single epoch"""
403 start = time.time()
404 total_tokens = 0
405 total_loss = 0
406 tokens = 0
407 n_accum = 0
408 for i, batch in enumerate(data_iter):
409 out = model.forward(
410 batch.src, batch.tgt, batch.src_mask, batch.tgt_mask
411 )
412 loss, loss_node = loss_compute(out, batch.tgt_y, batch.ntokens)
413 # loss_node = loss_node / accum_iter
414 if mode == "train" or mode == "train+log":
415 loss_node.backward()
416 train_state.step += 1
417 train_state.samples += batch.src.shape[0]
418 train_state.tokens += batch.ntokens
419 if i % accum_iter == 0:
420 optimizer.step()
421 optimizer.zero_grad(set_to_none=True)
422 n_accum += 1
423 train_state.accum_step += 1
424 scheduler.step()
425
426 total_loss += loss
427 total_tokens += batch.ntokens
428 tokens += batch.ntokens
429 if i % 40 == 1 and (mode == "train" or mode == "train+log"):
430 lr = optimizer.param_groups[0]["lr"]
431 elapsed = time.time() - start
432 print(
433 (
434 "Epoch Step: %6d | Accumulation Step: %3d | Loss: %6.2f "
435 + "| Tokens / Sec: %7.1f | Learning Rate: %6.1e"
436 )
437 % (i, n_accum, loss / batch.ntokens, tokens / elapsed, lr)
438 )
439 start = time.time()
440 tokens = 0
441 del loss
442 del loss_node
443 return total_loss / total_tokens, train_state
